func create_simple_nav_polygon(region: NavigationRegion2D, points_km: Array[Vector2], inflate: float = 0.15) -> void:
	"""
	Crée un NavigationPolygon simple à partir d'une liste de points (en km).
	inflate = marge en km autour du chemin (pour largeur de voie).
	"""
	var nav_poly = NavigationPolygon.new()
	var outline = PackedVector2Array()
	
	for p in points_km:
		outline.append(p * scale_factor)
	
	# Pour un vrai réseau, tu devras générer des polygones de largeur correcte
	# (offset du path). Ici version simplifiée pour démonstration.
	nav_poly.add_outline(outline)
	nav_poly.make_polygons_from_outlines()
	region.navigation_polygon = nav_poly


func load_road_graph_and_build_astar() -> AStar2D:
	"""
	Charge road_graph.json et construit un AStar2D prêt à l'emploi.
	"""
	var astar = AStar2D.new()
	var file = FileAccess.open("res://data/road_graph.json", FileAccess.READ)
	if not file:
		push_error("Impossible de charger road_graph.json")
		return astar
	
	var json = JSON.new()
	var err = json.parse(file.get_as_text())
	file.close()
	if err != OK:
		push_error("Erreur JSON road_graph")
		return astar
	
	var data = json.data
	var id_map = {}  # string id -> int id pour AStar
	
	# Ajouter les nodes
	var next_id = 0
	for node in data.nodes:
		var pos = Vector2(node.pos.x, node.pos.y) * scale_factor
		astar.add_point(next_id, pos)
		id_map[node.id] = next_id
		next_id += 1
	
	# Ajouter les edges
	for edge in data.edges:
		var from_id = id_map[edge.from]
		var to_id = id_map[edge.to]
		var cost = edge.length_km * scale_factor
		
		# Coût différent selon type de route (optionnel)
		match edge.type:
			"highway":
				cost *= 0.8
			"primary":
				cost *= 1.0
			"secondary":
				cost *= 1.3
			"local":
				cost *= 1.8
			"bridge":
				cost *= 1.1
		
		astar.connect_points(from_id, to_id, edge.get("bidirectional", true))
		# Note: AStar2D n'a pas de poids direct sur connect, 
		# utilise set_point_weight_scale si besoin plus fin.
	
	print("AStar2D construit avec ", astar.get_point_count(), " points")
	return astar


func create_path2d_from_transit(line_id: String, network: String = "train") -> Path2D:
	"""
	Crée un Path2D à partir de transit_paths.json
	"""
	var file = FileAccess.open("res://data/transit_paths.json", FileAccess.READ)
	if not file:
		return null
	
	var json = JSON.new()
	json.parse(file.get_as_text())
	file.close()
	
	var data = json.data
	if not data.has(network) or not data[network].has(line_id):
		push_error("Ligne introuvable : " + network + "/" + line_id)
		return null
	
	var line_data = data[network][line_id]
	var path = Path2D.new()
	path.name = line_id
	var curve = Curve2D.new()
	
	for p in line_data.points:
		curve.add_point(Vector2(p.x, p.y) * scale_factor)
	
	path.curve = curve
	return path
